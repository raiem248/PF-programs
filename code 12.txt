#include <iostream>
using namespace std;

int main() {
    int rows;
    cout << "Enter number of rows: ";
    cin >> rows;

    cout << "\nRight Triangle:\n";
    for (int i = 1; i <= rows; i++) {
        for (int j = 1; j <= i; j++)
            cout << "* ";
        cout << endl;
    }

    cout << "\nPyramid:\n";
    for (int i = 1; i <= rows; i++) {
        for (int s = 1; s <= rows - i; s++)
            cout << " ";
        for (int j = 1; j <= 2 * i - 1; j++)
            cout << "*";
        cout << endl;
    }
    return 0;
}
