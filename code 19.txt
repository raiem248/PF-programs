#include <iostream>
#include <string>
#include <cctype>
using namespace std;

int main() {
    string str;
    cout << "Enter a string (no spaces): ";
    cin >> str;

    string rev = "";
    for (int i = str.length() - 1; i >= 0; i--)
        rev += str[i];

    int vowels = 0;
    for (char ch : str) {
        ch = tolower(ch);
        if (ch == 'a' || ch == 'e' || ch == 'i' || ch == 'o' || ch == 'u')
            vowels++;
    }

    cout << "Length       : " << str.length() << endl;
    cout << "Reversed     : " << rev << endl;
    cout << "Vowels count : " << vowels << endl;
    cout << (str == rev ? "It is a palindrome." : "It is not a palindrome.") << endl;
    return 0;
}
