#include <iostream>
#include <string>
using namespace std;

struct Student {
    int rollNo;
    string name;
    float marks[3];
    float total;
    float percentage;
};

int main() {
    int n;
    Student s[50];
    cout << "How many students (max 50)? ";
    cin >> n;

    for (int i = 0; i < n; i++) {
        cout << "\nStudent " << i + 1 << endl;
        cout << "Roll No: ";
        cin >> s[i].rollNo;
        cout << "Name: ";
        cin >> s[i].name;
        s[i].total = 0;
        for (int j = 0; j < 3; j++) {
            cout << "Marks in subject " << j + 1 << ": ";
            cin >> s[i].marks[j];
            s[i].total += s[i].marks[j];
        }
        s[i].percentage = s[i].total / 3;
    }

    cout << "\nRoll No\tName\t\tTotal\tPercentage\n";
    cout << "----------------------------------------------\n";
    for (int i = 0; i < n; i++)
        cout << s[i].rollNo << "\t" << s[i].name << "\t\t"
             << s[i].total << "\t" << s[i].percentage << "%" << endl;
    return 0;
}
